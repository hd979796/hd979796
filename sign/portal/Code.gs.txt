/**
 * 박문철보험대리점 업무포탈 — 백엔드 (Google Apps Script 웹앱)
 * 데이터는 이 스크립트가 연결된 구글 시트에 탭(컬렉션)별로 저장된다.
 *
 * 설치 순서는 README(설치안내.md) 참고.
 * 스크립트 속성(프로젝트 설정 > 스크립트 속성):
 *   PASSCODE          포탈 비밀번호 (필수)
 *   ANTHROPIC_API_KEY 통화 정리(AI 요약)용 키 (선택, 없으면 정리 기능만 꺼짐)
 *   MODEL             기본 claude-opus-5-5
 */

var COLS = ["cases", "requests", "templates", "notes", "calls", "todos", "profiles", "customers", "customers_meta", "portal"];
var JSON_FIELDS = ["family", "todos", "support", "contracts", "renewal", "missing"]; // 배열·객체로 다루는 칸
var NUM_FIELDS = ["premium", "prevPremium", "order", "portalNo", "all", "long", "gen", "transfer", "count", "total"]; // 숫자로 되돌릴 칸 (customers.car 건수 포함)

function doGet(e) {
  var p = (e && e.parameter) || {};
  /* 주소만 열면(action 없음) 화면을 보여준다. ?page=sign 동의서 서명, ?page=game 우당탕탕, 그 외 업무포탈 */
  if (!p.action) {
    var PAGES = { portal: "박문철보험대리점 업무포탈", sign: "동의서 서명", game: "우당탕탕 요절복통" };
    var pg = PAGES[p.page] ? p.page : "portal";
    return HtmlService.createHtmlOutputFromFile(pg)
      .setTitle(PAGES[pg])
      .addMetaTag("viewport", "width=device-width,initial-scale=1,viewport-fit=cover")
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }
  if (p.action === "ping") return out({ ok: true, name: "박문철보험대리점 업무포탈", time: new Date().toISOString(), ai: !!prop("ANTHROPIC_API_KEY") });
  if (!auth(p.key)) return out({ ok: false, error: "unauthorized" });
  if (p.action === "recs") return out({ ok: true, data: listRecs(Number(p.days) || 7) });
  if (p.action === "rec") return out(recFile(p.id));
  if (p.action === "all") {
    var res = {};
    COLS.forEach(function (c) { if (c !== "customers") res[c] = readAll(c); });
    return out({ ok: true, data: res, time: new Date().toISOString() });
  }
  if (p.action === "list") return out({ ok: true, data: readAll(p.col) });
  if (p.action === "get") return out({ ok: true, data: readOne(p.col, p.id) });
  return out({ ok: false, error: "unknown action" });
}

function doPost(e) {
  var body = {};
  try { body = JSON.parse(e.postData.contents || "{}"); } catch (err) { return out({ ok: false, error: "bad json" }); }
  if (!auth(body.key)) return out({ ok: false, error: "unauthorized" });
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000);
    switch (body.action) {
      case "set": return out({ ok: true, data: writeDoc(body.col, body.id, body.data, false) });
      case "update": return out({ ok: true, data: writeDoc(body.col, body.id, body.data, true) });
      case "delete": deleteDoc(body.col, body.id); return out({ ok: true });
      case "batch":
        (body.writes || []).forEach(function (w) {
          if (w.op === "delete") deleteDoc(w.col, w.id); else writeDoc(w.col, w.id, w.data, w.op === "update");
        });
        return out({ ok: true, count: (body.writes || []).length });
      case "replace_customers": replaceCustomers(body.rows || [], body.meta || {}); return out({ ok: true, count: (body.rows || []).length });
      case "summarize": return out(summarize(body));
      default: return out({ ok: false, error: "unknown action" });
    }
  } catch (err) {
    return out({ ok: false, error: String(err && err.message || err) });
  } finally {
    try { lock.releaseLock(); } catch (e2) {}
  }
}

/* ---------- 시트 접근 ---------- */
var SHEET_ID = "1lab_bOLUQ02S1SlXG_wIWMoejj0wnbLyNl89U1qrv-w"; // 업무포탈 데이터 시트
function ss() { try { var a = SpreadsheetApp.getActiveSpreadsheet(); if (a) return a; } catch (e) {} return SpreadsheetApp.openById(SHEET_ID); }
function prop(k) { return PropertiesService.getScriptProperties().getProperty(k) || ""; }
/* 비밀번호 확인. 틀린 시도가 15분 안에 5번 쌓이면 15분 동안 전부 거절한다(짧은 비밀번호 보호). */
function auth(key) {
  var pc = prop("PASSCODE"); if (!pc) return false;
  var cache = CacheService.getScriptCache(); var fails = Number(cache.get("auth_fails") || 0);
  if (fails >= 5) return false;
  if (String(key) === pc) return true;
  cache.put("auth_fails", String(fails + 1), 900);
  return false;
}
function out(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }

function sheet(col) {
  if (COLS.indexOf(col) < 0) throw new Error("없는 컬렉션: " + col);
  var sh = ss().getSheetByName(col);
  if (!sh) { sh = ss().insertSheet(col); textFormat(sh); sh.appendRow(["id", "updatedAt"]); sh.setFrozenRows(1); }
  return sh;
}
/* 모든 칸을 '일반 텍스트'로 두어 날짜·전화번호·차량번호가 변형되지 않게 한다 */
function textFormat(sh) { sh.getRange(1, 1, sh.getMaxRows(), sh.getMaxColumns()).setNumberFormat("@"); }
function headers(sh) { var lc = sh.getLastColumn(); return lc ? sh.getRange(1, 1, 1, lc).getValues()[0].map(String) : []; }
function encode(v) {
  if (v === null || v === undefined) return "";
  if (typeof v === "object") return JSON.stringify(v);
  return v;
}
function decode(k, v) {
  if (v === "" || v === null || v === undefined) return undefined;
  if (v instanceof Date) return Utilities.formatDate(v, "Asia/Seoul", "yyyy-MM-dd");
  if (typeof v === "string") {
    if (JSON_FIELDS.indexOf(k) >= 0 || /^[\[{]/.test(v)) { try { return JSON.parse(v); } catch (e) { return v; } }
    if (v === "TRUE" || v === "true") return true;
    if (v === "FALSE" || v === "false") return false;
    if (NUM_FIELDS.indexOf(k) >= 0 && /^-?\d+(\.\d+)?$/.test(v)) return Number(v);
    if (k === "car" && /^\d+$/.test(v)) return Number(v); // customers 탭의 자동차 건수
  }
  return v;
}
function rowToDoc(h, row) {
  var d = {};
  for (var i = 0; i < h.length; i++) { var v = decode(h[i], row[i]); if (v !== undefined) d[h[i]] = v; }
  return d;
}
function readAll(col) {
  var sh = sheet(col); var lr = sh.getLastRow(); if (lr < 2) return [];
  var h = headers(sh); var rows = sh.getRange(2, 1, lr - 1, h.length).getValues();
  return rows.filter(function (r) { return r[0] !== ""; }).map(function (r) { return rowToDoc(h, r); });
}
function findRow(sh, id) {
  var lr = sh.getLastRow(); if (lr < 2) return -1;
  var ids = sh.getRange(2, 1, lr - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) if (String(ids[i][0]) === String(id)) return i + 2;
  return -1;
}
function readOne(col, id) { var sh = sheet(col); var r = findRow(sh, id); if (r < 0) return null; var h = headers(sh); return rowToDoc(h, sh.getRange(r, 1, 1, h.length).getValues()[0]); }
function writeDoc(col, id, data, merge) {
  if (!id) throw new Error("id 필요");
  var sh = sheet(col); var h = headers(sh); var r = findRow(sh, id);
  var cur = {};
  if (r > 0 && merge) cur = rowToDoc(h, sh.getRange(r, 1, 1, h.length).getValues()[0]);
  var doc = merge ? Object.assign({}, cur, data || {}) : Object.assign({}, data || {});
  doc.id = id; doc.updatedAt = new Date().toISOString();
  Object.keys(doc).forEach(function (k) { if (h.indexOf(k) < 0) { h.push(k); sh.getRange(1, h.length).setValue(k); } });
  var row = h.map(function (k) { return encode(doc[k]); });
  if (r < 0) { r = sh.getLastRow() + 1; if (r > sh.getMaxRows()) sh.insertRowsAfter(sh.getMaxRows(), 100); }
  sh.getRange(r, 1, 1, row.length).setNumberFormat("@").setValues([row]);
  return doc;
}
function deleteDoc(col, id) { var sh = sheet(col); var r = findRow(sh, id); if (r > 0) sh.deleteRow(r); }
function replaceCustomers(rows, meta) {
  var sh = sheet("customers"); sh.clearContents(); if (sh.getMaxRows() < rows.length + 2) sh.insertRowsAfter(sh.getMaxRows(), rows.length + 2 - sh.getMaxRows()); textFormat(sh);
  var keys = ["name", "birth", "sex", "type", "all", "long", "car", "gen", "transfer", "privacyEnd", "marketingEnd", "job", "addr"];
  var values = [keys].concat(rows.map(function (r) { return keys.map(function (k) { return r[k] === undefined ? "" : r[k]; }); }));
  sh.getRange(1, 1, values.length, keys.length).setValues(values); sh.setFrozenRows(1);
  writeDoc("customers_meta", "meta", Object.assign({ total: rows.length }, meta), false);
}

/* ---------- 통화 정리 (Claude API) ---------- */
function summarize(body) {
  var key = prop("ANTHROPIC_API_KEY");
  if (!key) return { ok: false, error: "no_ai", message: "ANTHROPIC_API_KEY가 설정되지 않았습니다" };
  var model = prop("MODEL") || "claude-opus-5-5";
  var payload = {
    model: model, max_tokens: 4000,
    fallbacks: "default",
    system: body.system || "",
    messages: [{ role: "user", content: String(body.prompt || "") }]
  };
  var res = UrlFetchApp.fetch("https://api.anthropic.com/v1/messages", {
    method: "post", contentType: "application/json", muteHttpExceptions: true,
    headers: { "x-api-key": key, "anthropic-version": "2023-06-01", "anthropic-beta": "server-side-fallback-2026-07-01" },
    payload: JSON.stringify(payload)
  });
  var code = res.getResponseCode(); var txt = res.getContentText();
  if (code !== 200) return { ok: false, error: "api_" + code, message: txt.slice(0, 300) };
  var j = JSON.parse(txt);
  if (j.stop_reason === "refusal") return { ok: false, error: "refusal", message: "모델이 응답을 거부했습니다" };
  var text = (j.content || []).filter(function (b) { return b.type === "text"; }).map(function (b) { return b.text; }).join("");
  var m = text.match(/\{[\s\S]*\}/);
  var data = null; try { data = JSON.parse(m ? m[0] : text); } catch (e) {}
  return { ok: true, text: text, data: data };
}

/** seed.json을 올린 뒤 그 주소(사이트 주소/seed.json 또는 Drive 파일 id)로 초기 데이터를 넣는다. 한 번만 실행. */
function importSeed(urlOrFileId) {
  var txt;
  if (/^https?:\/\//.test(urlOrFileId)) txt = UrlFetchApp.fetch(urlOrFileId).getContentText();
  else txt = DriveApp.getFileById(urlOrFileId).getBlob().getDataAsString("UTF-8");
  var seed = JSON.parse(txt);
  ["cases", "requests", "templates", "notes", "calls", "todos", "profiles", "portal"].forEach(function (c) {
    var docs = seed[c] || {}; var ids = Object.keys(docs); if (!ids.length) return;
    var sh = sheet(c); sh.clearContents(); var h = ["id", "updatedAt"];
    if (sh.getMaxRows() < ids.length + 2) sh.insertRowsAfter(sh.getMaxRows(), ids.length + 2 - sh.getMaxRows()); textFormat(sh);
    ids.forEach(function (id) { Object.keys(docs[id]).forEach(function (k) { if (h.indexOf(k) < 0) h.push(k); }); });
    var values = [h].concat(ids.map(function (id) { var d = docs[id]; return h.map(function (k) { return k === "id" ? id : encode(d[k]); }); }));
    sh.getRange(1, 1, values.length, h.length).setValues(values); sh.setFrozenRows(1);
    Logger.log(c + ": " + ids.length + "건");
  });
  if (seed.customers) { replaceCustomers(seed.customers, (seed.customers_meta || {}).meta || {}); Logger.log("customers: " + seed.customers.length + "명"); }
}

/* ---------- 업무폰 통화 녹음 (PC 자동 정리 프로그램이 사용: ai 오피스/통화자동정리) ---------- */
var REC_FOLDERS = { "17zSkwfeDR-0JpB2bUu9lMSry8THAAN7A": "hwang", "1uuh44lzLXzbynv0y1TTam6tGXJZtXlSB": "park" };
/** 아직 통화 기록(calls.files)에 없는 녹음 파일 목록 */
function listRecs(days) {
  var done = {};
  readAll("calls").forEach(function (c) {
    var f = c.files; if (typeof f === "string") { try { f = JSON.parse(f); } catch (e) { f = [f]; } }
    (f || []).forEach(function (x) { done[x] = 1; });
  });
  var sinceD = new Date(Date.now() - days * 864e5);
  var since = Utilities.formatDate(sinceD, "UTC", "yyyy-MM-dd'T'HH:mm:ss");
  var res = [];
  Object.keys(REC_FOLDERS).forEach(function (fid) {
    recSearch_(fid, since).forEach(function (f) {
      if (done[f.getId()] || !/\.(m4a|mp3|amr|3gp|wav|aac|ogg)$/i.test(f.getName())) return;
      var cr = f.getDateCreated(); if (cr < sinceD) return;
      res.push({ id: f.getId(), name: f.getName(), room: REC_FOLDERS[fid], size: f.getSize(), created: cr.toISOString() });
    });
  });
  res.sort(function (a, b) { return a.created < b.created ? -1 : 1; });
  return res;
}
/** 녹음 파일 내용(base64). 녹음 폴더 안의 파일만 내준다. */
function recFile(id) {
  var f = DriveApp.getFileById(id); var ok = false; var ps = f.getParents();
  while (ps.hasNext()) { if (REC_FOLDERS[ps.next().getId()]) ok = true; }
  if (!ok) return { ok: false, error: "not_rec" };
  return { ok: true, name: f.getName(), b64: Utilities.base64Encode(f.getBlob().getBytes()) };
}

/** 드라이브 검색 문법이 v2/v3 어느 쪽이든 동작하도록 차례로 시도 */
function recSearch_(fid, since) {
  var base = "'" + fid + "' in parents and trashed = false";
  var qs = [base + " and createdTime > '" + since + "'", base + " and createdDate > '" + since + "'", base];
  for (var i = 0; i < qs.length; i++) {
    try { var it = DriveApp.searchFiles(qs[i]); var arr = []; while (it.hasNext()) arr.push(it.next()); return arr; } catch (e) {}
  }
  return [];
}
